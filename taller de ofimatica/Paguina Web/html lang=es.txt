  <script>
    // Variables de Estado de la Aplicación
    let timeoutBusqueda = null;
    let AlumnoValido = false;

    // Inicialización del sistema al cargar el DOM
    document.addEventListener("DOMContentLoaded", function() {
      // Inicializar Google Charts
      google.charts.load('current', {'packages':['corechart', 'bar']});
      google.charts.setOnLoadCallback(inicializarComponentes);
      
      // Enlazar evento debounce para clave de alumno
      document.getElementById("claveAlumno").addEventListener("input", manejarCambioClave);
    });

    /**
     * Coordina la carga inicial sincronizada de datos de la API.
     */
    function inicializarComponentes() {
      cargarMateriales();
      actualizarTablaPrestamos();
    }

    /**
     * Muestra notificaciones flotantes elegantes (Toasts).
     */
    function showToast(message, type = 'info') {
      const container = document.getElementById("toastContainer");
      const toast = document.createElement("div");
      toast.className = `fade-in flex items-center p-4 rounded-lg shadow-md border text-sm max-w-xs transition-all duration-300 transform translate-y-0`;
      
      if (type === 'success') {
        toast.className += " bg-emerald-50 text-emerald-800 border-emerald-200";
      } else if (type === 'error') {
        toast.className += " bg-rose-50 text-rose-800 border-rose-200";
      } else {
        toast.className += " bg-blue-50 text-blue-800 border-blue-200";
      }
      
      toast.innerHTML = `<span>${message}</span>`;
      container.appendChild(toast);
      
      // Auto destruirse a los 4 segundos
      setTimeout(() => {
        toast.style.opacity = '0';
        setTimeout(() => toast.remove(), 300);
      }, 4000);
    }

    function toggleSpinner(show) {
      document.getElementById("loadingOverlay").classList.toggle("active", show);
    }

    /**
     * Controla el evento input mediante un Debounce de 400ms 
     * para no saturar con llamadas consecutivas en cada tecla.
     */
    function manejarCambioClave(e) {
      clearTimeout(timeoutBusqueda);
      const clave = e.target.value.trim();
      const inputNombre = document.getElementById("nombreAlumno");
      const spinner = document.getElementById("searchingAlumnoSpinner");
      
      if (!clave) {
        inputNombre.value = "";
        evaluarEstadoBotonForm();
        return;
      }

      spinner.classList.remove("hidden");
      
      timeoutBusqueda = setTimeout(() => {
        google.script.run
          .withSuccessHandler(function(nombre) {
            spinner.classList.add("hidden");
            if (nombre) {
              inputNombre.value = nombre;
              AlumnoValido = true;
              showToast("Alumno verificado correctamente.", "success");
            } else {
              inputNombre.value = "Alumno No Encontrado";
              AlumnoValido = false;
            }
            evaluarEstadoBotonForm();
          })
          .withFailureHandler(function() {
            spinner.classList.add("hidden");
            inputNombre.value = "Error al conectar base de datos";
            AlumnoValido = false;
            evaluarEstadoBotonForm();
          })
          .buscarNombrePorClave(clave);
      }, 400);
    }

    function evaluarEstadoBotonForm() {
      const selectMat = document.getElementById("selectMaterial");
      const btn = document.getElementById("btnSubmit");
      
      if (AlumnoValido && selectMat.value !== "" && selectMat.value !== "cargando") {
        btn.disabled = false;
      } else {
        btn.disabled = true;
      }
    }

    /**
     * Consume la API del backend para poblar el dropdown de materiales.
     */
    function cargarMateriales() {
      google.script.run
        .withSuccessHandler(function(materiales) {
          const select = document.getElementById("selectMaterial");
          select.innerHTML = '<option value="" disabled selected>-- Elige un material --</option>';
          
          if (materiales.length === 0) {
            select.innerHTML = '<option value="" disabled>No hay insumos disponibles</option>';
          } else {
            materiales.forEach(m => {
              let opt = document.createElement("option");
              opt.value = m["ID"];
              opt.textContent = `${m["Nombre"]} (${m["Cantidad Disponible"]} disp.)`;
              select.appendChild(opt);
            });
          }
          select.addEventListener("change", evaluarEstadoBotonForm);
          evaluarEstadoBotonForm();
          renderizarGrafico(); // Renderizar gráfico una vez se carguen o muten datos
        })
        .getMaterialesDisponibles();
    }

    /**
     * Consulta y dibuja las métricas de inventario mediante Google Charts.
     */
    function renderizarGrafico() {
      google.script.run
        .withSuccessHandler(function(datos) {
          if (!datos || datos.length <= 1) {
            document.getElementById("chartInventario").innerHTML = "Sin datos suficientes para procesar métricas.";
            return;
          }
          
          const dataTable = google.visualization.arrayToDataTable(datos);
          const options = {
            isStacked: true,
            colors: ['#2563eb', '#f59e0b'], // Esis de colores Tailwind (Blue-600 y Amber-500)
            chartArea: { width: '80%', height: '75%' },
            legend: { position: 'top', alignment: 'center' },
            vAxis: { minValue: 0, textStyle: { fontSize: 11 } },
            hAxis: { textStyle: { fontSize: 11 } }
          };
          
          const chart = new google.visualization.ColumnChart(document.getElementById('chartInventario'));
          chart.draw(dataTable, options);
        })
        .getDatosGraficoInventario();
    }

    /**
     * Envía la solicitud de inserción al backend.
     */
    function submitPrestamo() {
      const data = {
        claveAlumno: document.getElementById("claveAlumno").value.trim(),
        idMaterial: document.getElementById("selectMaterial").value,
        cantidad: parseInt(document.getElementById("cantidadPrestamo").value)
      };

      toggleSpinner(true);

      google.script.run
        .withSuccessHandler(function(res) {
          toggleSpinner(false);
          if (res.success) {
            showToast(res.message, "success");
            // Resetear el formulario de manera controlada
            document.getElementById("formPrestamo").reset();
            document.getElementById("nombreAlumno").value = "";
            AlumnoValido = false;
            evaluarEstadoBotonForm();
            
            // Refrescar vistas asíncronas
            inicializarComponentes();
          } else {
            showToast(res.message, "error");
          }
        })
        .withFailureHandler(function(err) {
          toggleSpinner(false);
          showToast("Error crítico de red: " + err.message, "error");
        })
        .registrarPrestamo(data);
    }

    /**
     * Actualiza e inyecta dinámicamente las filas de los préstamos en estado activo.
     */
    function actualizarTablaPrestamos() {
      google.script.run
        .withSuccessHandler(function(prestamos) {
          const tbody = document.getElementById("tablaPrestamosBody");
          const contador = document.getElementById("contadorPrestamos");
          tbody.innerHTML = "";
          
          contador.textContent = `${prestamos.length} Activos`;
          
          if (prestamos.length === 0) {
            tbody.innerHTML = `
              <tr>
                <td colspan="6" class="px-6 py-8 text-center text-gray-400 font-medium">
                  No existen préstamos activos registrados actualmente.
                </td>
              </tr>`;
            return;
          }

          prestamos.forEach(p => {
            let tr = document.createElement("tr");
            tr.className = "hover:bg-gray-50 transition fade-in";
            tr.innerHTML = `
              <td class="px-6 py-4 whitespace-nowrap text-xs text-gray-500 font-mono">${p["Timestamp"]}</td>
              <td class="px-6 py-4 whitespace-nowrap">
                <div class="text-sm font-semibold text-gray-900">${p["Nombre Alumno"]}</div>
                <div class="text-xs text-gray-400 font-mono">Clave: ${p["Clave Alumno"]}</div>
              </td>
              <td class="px-6 py-4 whitespace-nowrap">
                <div class="text-sm text-gray-900 font-medium">${p["Nombre Material"]}</div>
                <div class="text-xs text-gray-400 font-mono">ID: ${p["ID Material"]}</div>
              </td>
              <td class="px-6 py-4 whitespace-nowrap text-center text-sm font-bold text-gray-700 bg-gray-50/50">${p["Cantidad"]}</td>
              <td class="px-6 py-4 whitespace-nowrap">
                <span class="px-2.5 py-1 text-xs font-bold rounded-full bg-amber-50 text-amber-700 border border-amber-200 inline-flex items-center">
                  <span class="w-1.5 h-1.5 rounded-full bg-amber-500 mr-1.5 animate-pulse"></span>
                  Pendiente
                </span>
              </td>
              <td class="px-6 py-4 whitespace-nowrap text-right text-sm font-medium">
                <button onclick="procesarDevolucion(${p.filaId})" 
                        class="bg-white text-gray-700 border border-gray-300 hover:bg-emerald-50 hover:text-emerald-700 hover:border-emerald-300 px-3 py-1.5 rounded-md transition shadow-sm font-medium text-xs uppercase tracking-wider">
                  Devolver
                </button>
              </td>
            `;
            tbody.appendChild(tr);
          });
        })
        .getPrestamosActivos();
    }

    /**
     * Ejecuta la acción de devolución de un material por su ID de fila.
     */
    function procesarDevolucion(filaId) {
      if (!confirm("¿Confirmar recepción y cierre del préstamo seleccionado?")) return;
      
      toggleSpinner(true);
      
      google.script.run
        .withSuccessHandler(function(res) {
          toggleSpinner(false);
          if (res.success) {
            showToast(res.message, "success");
            inicializarComponentes();
          } else {
            showToast(res.message, "error");
          }
        })
        .withFailureHandler(function(err) {
          toggleSpinner(false);
          showToast("Error al procesar devolución: " + err.message, "error");
        })
        .registrarDevolucion(filaId);
    }
  </script>
